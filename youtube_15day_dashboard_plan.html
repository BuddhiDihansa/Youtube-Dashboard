Phase 3 Streamlit App (Days 10–12)
Day10
App structure + KPI cards
Layout, sidebar filters, metrics
App
✓
app.py skeleton
import streamlit as st import pandas as pd import plotly.express as px st.set_page_config(page_title="YouTube Creator Analytics", page_icon="🎥", layout="wide") st.title("🎥 Global YouTube Creator Analytics") st.caption("995 top creators · 2023 data · 28 metrics") df = pd.read_csv('yt_clean.csv') # Sidebar st.sidebar.header("Filters") cats = ['All'] + sorted(df['category'].dropna().unique()) selected = st.sidebar.selectbox("Category", cats) countries = st.sidebar.multiselect("Countries", df['Country'].unique()) if selected != 'All': df = df[df['category'] == selected] if countries: df = df[df['Country'].isin(countries)] # KPI row c1,c2,c3,c4 = st.columns(4) c1.metric("Total Channels", len(df)) c2.metric("Avg Subscribers", f"{df['subscribers'].mean()/1e6:.1f}M") c3.metric("Avg Yearly Earnings", f"${df['avg_earn'].mean():,.0f}") c4.metric("Countries", df['Country'].nunique())
